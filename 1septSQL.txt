-- Customers who placed an order
SELECT CustomerID, FirstName,LastName
FROM customer
WHERE CustomerID IN (
    SELECT DISTINCT CustomerID 
    FROM order_details
);

-- count orders per customer
SELECT CustomerID, FirstName,LastName, 
    (SELECT COUNT(*) 
     FROM order_details as o 
     WHERE o.CustomerID = c.CustomerID) AS total_orders
FROM Customer as c;

-- total price of all orders
SELECT (SELECT SUM(total_amount) 
FROM order_details) AS total_price;  

-- product which had no sales
SELECT ProductID,ProductName 
FROM products 
WHERE ProductID NOT IN 
(SELECT DISTINCT ProductID
 FROM order_details);  
 
 -- inner join-- returns customer records who placed order
SELECT 
    c.CustomerID,c.FirstName,o.order_id,o.total_amount
FROM Customer c
INNER JOIN order_details o 
    ON c.CustomerID = o.CustomerID;
    
-- left join  -- returns all the customers 
-- with orderid & amount. 
-- Those who didnt place order, details are null
SELECT c.FirstName, o.order_id, o.total_amount
FROM Customer as c
LEFT JOIN order_details as o 
ON c.CustomerID = o.CustomerID;

-- right join  
-- returns all details  from order table
-- & matching details in customer table
SELECT  o.order_id,c.FirstName,o.total_amount
FROM Customer as c
RIGHT JOIN order_details as o 
    ON c.CustomerID = o.CustomerID;
    
-- full outer join -- returns all details from both tables with matching records
SELECT c.CustomerID,c.FirstName, o.order_id,o.total_amount
FROM  Customer as c 
LEFT JOIN order_details as o 
ON c.CustomerID=o.CustomerID 
UNION
SELECT c.CustomerID,c.FirstName, o.order_id,o.total_amount
FROM  Customer as c 
RIGHT JOIN order_details as o  
ON c.CustomerID=o.CustomerID;

-- self join -- find repeat customers
SELECT 
    o1.CustomerID,
    o1.order_id AS initial_order,
    o1.order_date AS initial_date,
    o2.order_id AS next_order,
    o2.order_date AS next_date
FROM order_details as o1
INNER JOIN order_details as o2 
    ON o1.CustomerID = o2.CustomerID
    AND o1.order_id < o2.order_id
ORDER BY o1.CustomerID;

-- self join -- customer from same state
SELECT 
    c1.FirstName AS customer_a,
    c2.FirstName AS customer_b,
    c1.State
FROM customer c1
INNER JOIN customer c2 
    ON c1.State = c2.State 
    AND c1.CustomerID < c2.CustomerID;
    
-- cross join 
-- generate every possible combination of customers and orders
SELECT 
    c.CustomerID, c.FirstName, o.order_id
FROM Customer c
CROSS JOIN order_details o;
